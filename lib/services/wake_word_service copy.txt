
import 'dart:async';
import 'package:speech_to_text/speech_to_text.dart' as stt;
import 'package:flutter/foundation.dart';

/// WakeWordService handles wake word detection and question listening.
class WakeWordService {
  static bool wakeWordShouldBeActive = true; // Global flag to control listening
  static int _globalSessionId = 0;
  int _currentSessionId = 0;
  /// Optional callback to determine if wake word auto-restart is allowed (set by main app)
  bool Function()? shouldAllowWakeWordRestart;
  final stt.SpeechToText _wakeWordSpeech = stt.SpeechToText();
  final stt.SpeechToText _questionSpeech = stt.SpeechToText();
  bool _isListening = false;
  bool _wakeWordDetected = false;
  List<String> _wakeWords = ['hey brady']; // Default, can be set from settings
  Future<void> Function(String)? onWakeWord;
  Future<void> Function(String)? onQuestion;

  WakeWordService({List<String>? wakeWords, String? wakeWord}) {
    if (wakeWords != null && wakeWords.isNotEmpty) {
      _wakeWords = wakeWords.map((w) => w.toLowerCase()).toList();
    } else if (wakeWord != null) {
      _wakeWords = [wakeWord.toLowerCase()];
    }
  }

  /// Stops wake word listening only (does not affect question listening)
  Future<void> stopWakeWordListening() async {
    wakeWordShouldBeActive = false;
    debugPrint('[WakeWordService] stopWakeWordListening called.');
    _isListening = false;
    await _wakeWordSpeech.stop();
  }

  Future<bool> init() async {
    // Initialize both recognizers for completeness
    bool wakeWordInit = await _wakeWordSpeech.initialize(
      onStatus: (status) {
        debugPrint('[WakeWordService] (init) Speech status: $status');
      },
      onError: (error) {
        debugPrint('[WakeWordService] (init) Speech error: $error');
      },
    );
    bool questionInit = await _questionSpeech.initialize(
      onStatus: (status) {
        debugPrint('[QuestionService] (init) Speech status: $status');
      },
      onError: (error) {
        debugPrint('[QuestionService] (init) Speech error: $error');
      },
    );
    return wakeWordInit && questionInit;
  }

  Future<void> startWakeWordListening() async {
    // Force _isListening to false and stop any previous recognizer before starting
    _isListening = false;
    await _wakeWordSpeech.stop();
    await Future.delayed(const Duration(milliseconds: 400));
    if (!wakeWordShouldBeActive) {
      debugPrint('[WakeWordService] startWakeWordListening: ABORTED because wakeWordShouldBeActive is false, sessionId=$_currentSessionId');
      return;
    }
    _globalSessionId++;
    _currentSessionId = _globalSessionId;
    debugPrint('[WakeWordService] startWakeWordListening called. sessionId=$_currentSessionId');
    if (_isListening) return;
    _isListening = true;
    _wakeWordDetected = false;
    debugPrint('[WakeWordService] Initializing speech-to-text for wake word listening...');
    final thisSessionId = _currentSessionId;
    bool available = await _wakeWordSpeech.initialize(
      onStatus: (status) {
        debugPrint('[WakeWordService] Speech status: $status');
        debugPrint('[WakeWordService] onStatus callback. sessionId=$thisSessionId, currentSessionId=$_currentSessionId');
        if (thisSessionId != _currentSessionId) {
          debugPrint('[WakeWordService] onStatus callback: ignoring stale sessionId=$thisSessionId');
          return;
        }
        // Auto-restart if recognizer is done and not in question mode
        if (status == 'done' && !_wakeWordDetected) {
          if (!wakeWordShouldBeActive) {
            debugPrint('[WakeWordService] Wake word recognizer done. Auto-restarting? false (wakeWordShouldBeActive is false), sessionId=$thisSessionId');
            _isListening = false;
            return;
          }
          final allowRestart = shouldAllowWakeWordRestart == null ? true : shouldAllowWakeWordRestart!();
          debugPrint('[WakeWordService] Wake word recognizer done. Auto-restarting? $allowRestart, sessionId=$thisSessionId');
          _isListening = false;
          if (allowRestart) {
            Future.delayed(const Duration(milliseconds: 500), startWakeWordListening);
          } else {
            debugPrint('[WakeWordService] Auto-restart prevented by shouldAllowWakeWordRestart callback.');
          }
        } else if (status == 'done' && _wakeWordDetected) {
          debugPrint('[WakeWordService] Wake word recognizer done after detection. Not auto-restarting.');
        }
      },
      onError: (error) {
        debugPrint('Speech error: $error');
        debugPrint('[WakeWordService] onError callback. sessionId=$thisSessionId, currentSessionId=$_currentSessionId');
        if (thisSessionId != _currentSessionId) {
          debugPrint('[WakeWordService] onError callback: ignoring stale sessionId=$thisSessionId');
          return;
        }
      },
    );
    if (!available) {
      debugPrint('[WakeWordService] ERROR: Speech-to-text initialization failed. Wake word listening not started.');
      _isListening = false;
      return;
    }
    debugPrint('[WakeWordService] Speech-to-text initialized. Listening for wake word(s): ${_wakeWords.join(', ')}');
    await _wakeWordSpeech.listen(
      onResult: (result) async {
        final transcript = result.recognizedWords.toLowerCase();
        debugPrint('[WakeWordService] Wake word listen transcript: ${transcript}');
        if (!_wakeWordDetected && _wakeWords.any((w) => transcript.contains(w))) {
          _wakeWordDetected = true;
          debugPrint('[WakeWordService] Wake word detected in transcript!');
          _wakeWordSpeech.stop();
          await Future.delayed(const Duration(milliseconds: 300));
          if (onWakeWord != null) {
            await onWakeWord!(transcript);
          }
          // Do NOT call startQuestionListening here; let the main app handle it after TTS.
        }
      },
      listenFor: const Duration(minutes: 5),
      pauseFor: const Duration(seconds: 3),
      partialResults: true,
      localeId: 'en_US',
      cancelOnError: true,
    );
  }

  Future<void> startQuestionListening({int retryCount = 0}) async {
    if (!_wakeWordDetected) return;
    debugPrint('[QuestionService] Forcing _isListening = false before question recognizer.');
    _isListening = false;
    debugPrint('[QuestionService] Stopping any previous recognizer before initializing for question...');
    await _questionSpeech.stop();
    await Future.delayed(const Duration(milliseconds: 400));
    debugPrint('[QuestionService] Initializing speech-to-text for question listening... (retryCount: $retryCount)');
    bool available = await _questionSpeech.initialize(
      onStatus: (status) {
        debugPrint('[QuestionService] Speech status (question): $status');
        if (status == 'done') {
          debugPrint('[QuestionService] Question recognizer done. Resetting wakeWordDetected to false.');
          _wakeWordDetected = false;
        }
      },
      onError: (error) async {
        debugPrint('[QuestionService] Speech error (question): $error');
        final msg = error.errorMsg.toLowerCase();
        if (retryCount < 2 && (msg.contains('no speech detected') || msg.contains('error_no_match'))) {
          debugPrint('[QuestionService] No speech detected, retrying question listening (attempt ${retryCount + 1})...');
          await Future.delayed(const Duration(milliseconds: 400));
          startQuestionListening(retryCount: retryCount + 1);
        } else {
          debugPrint('[QuestionService] Max retries reached or other error. Resetting wakeWordDetected to false.');
          _wakeWordDetected = false;
        }
      },
    );
    if (!available) {
      debugPrint('[QuestionService] ERROR: Speech-to-text initialization failed. Question listening not started.');
      _wakeWordDetected = false;
      return;
    }
    debugPrint('[QuestionService] Speech-to-text initialized. Listening for question...');
    await _questionSpeech.listen(
      onResult: (result) async {
        final question = result.recognizedWords.trim();
        debugPrint('[QuestionService] Question listen transcript: \\${question}');
        // Only trigger onQuestion for final results, not partials
        if (result.finalResult && question.isNotEmpty && onQuestion != null) {
          debugPrint('[QuestionService] Final question recognized, announcing confirmation and firing onQuestion callback.');
          // Announce confirmation before firing onQuestion
        if (onQuestion != null) {
          await onQuestion!(question);
        }
          // After handling the question, reset wakeWordDetected to false
          _wakeWordDetected = false;
        } else {
          debugPrint('[QuestionService] Partial result received, not firing onQuestion.');
        }
      },
      listenFor: const Duration(seconds: 15),
      pauseFor: const Duration(seconds: 3),
      partialResults: true,
      localeId: 'en_US',
      cancelOnError: true,
    );
  }

  void stop() {
    wakeWordShouldBeActive = false;
    _isListening = false;
    _wakeWordDetected = false;
    _currentSessionId = -1; // Invalidate all previous sessions
    _wakeWordSpeech.stop();
    _questionSpeech.stop();
  }

  bool get isListening => _isListening;
  bool get wakeWordDetected => _wakeWordDetected;
}
