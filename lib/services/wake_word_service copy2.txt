import 'dart:async';
import 'package:flutter/foundation.dart';
import 'package:speech_to_text/speech_to_text.dart';
import 'package:speech_to_text/speech_recognition_result.dart';

class WakeWordService {
  final List<String> wakeWords;
  static bool wakeWordShouldBeActive = true;

  // Callback hooks
  void Function(String transcript)? onWakeWord;
  void Function(String question)? onQuestion;
  bool Function()? shouldAllowWakeWordRestart;

  final SpeechToText _wakeWordSpeech = SpeechToText();
  final SpeechToText _questionSpeech = SpeechToText();
  bool _isListening = false;

  bool _wakeWordDetected = false;

  int _globalSessionId = 0;
  int _currentSessionId = 0;

  Timer? _questionTimeoutTimer;
  String _lastPartialQuestion = '';

  Completer<void>? _stopCompleter;

  bool _shouldRestartWakeWordListening = true; // Add this to your class
  WakeWordService({required this.wakeWords});

  /// Robust stop: returns when recognizer is fully stopped.
  Future<void> stopWakeWordListening() async {
    debugPrint('[WakeWordService] stopWakeWordListening called');
    debugPrint("[WakeWordService] Before _isListening: $_isListening");
    if (!_isListening) return;
    _isListening = false;
    _stopCompleter = Completer<void>();
    await _wakeWordSpeech.stop();
    await _stopCompleter!.future; // Only completes in onStatus when status == 'done'
    debugPrint("[WakeWordService] After _isListening: $_isListening");
    // Wait for onStatus: done or timeout
    return _stopCompleter!.future.timeout(
      const Duration(seconds: 2),
      onTimeout: () {
        debugPrint('[WakeWordService] stopWakeWordListening: timeout');
        if (!_stopCompleter!.isCompleted) _stopCompleter!.complete();
      },
    );
  }

  /// Start listening for the wake word, only after previous session is stopped.
  Future<void> startWakeWordListening() async {
    debugPrint('[WakeWordService] startWakeWordListening called');
    if (!wakeWordShouldBeActive) {
      debugPrint('[WakeWordService] startWakeWordListening: ABORTED (shouldBeActive is false)');
      return;
    }
    // Wait for any previous stop to finish
    if (_isListening) {
      await stopWakeWordListening();
      await Future.delayed(const Duration(milliseconds: 200));
    }
    _globalSessionId++;
    _currentSessionId = _globalSessionId;
    _isListening = true;
    debugPrint('Starting wake word listening. sessionId=$_currentSessionId.  wakeWordDetcted set to false');
    _wakeWordDetected = false;

    await _wakeWordSpeech.initialize(
      onStatus: (status) => _onStatus(status, _currentSessionId),
      onError: (error) => _onError(error, _currentSessionId),
    );
    await _wakeWordSpeech.listen(
      onResult: (result) => _onResult(result, _currentSessionId),
      listenMode: ListenMode.dictation,
      partialResults: true,
      cancelOnError: false,
    );
    debugPrint('[WakeWordService] Listening started. sessionId=$_currentSessionId');
  }


  Future<void> stopAllRecognizers() async {
    await stopWakeWordListening();
    debugPrint('[WakeWordService] stopAllRecognizers called. Stopping question recognizer.');
    _questionTimeoutTimer?.cancel();
    _lastPartialQuestion = '';
    //_wakeWordDetected = false;
    _stopCompleter = Completer<void>();
    debugPrint('[WakeWordService] Before _questionSpeech.stop()');
    await _questionSpeech.stop();
    debugPrint('[WakeWordService] After _questionSpeech.stop()');
    if (_stopCompleter != null && !_stopCompleter!.isCompleted) {
      _stopCompleter!.complete();
      debugPrint('[WakeWordService] stopAllRecognizers: _stopCompleter completed');
    } else {
      debugPrint('[WakeWordService] stopAllRecognizers: _stopCompleter was already completed');
    }
    debugPrint('[WakeWordService] All recognizers stopped.');
  }

  void _onStatus(String status, int sessionId) {
      debugPrint('[WakeWordService] onStatus callback. sessionId=$sessionId, currentSessionId=$_currentSessionId, status=$status');
      if (sessionId != _currentSessionId) {
        debugPrint('[WakeWordService] onStatus callback: ignoring stale sessionId=$sessionId');
        return;
      }
      if (status == 'done' || status == 'notListening') {
        _isListening = false;
        if (_stopCompleter != null && !_stopCompleter!.isCompleted) {
          _stopCompleter!.complete();
        }
      }
    }

  void pauseWakeWordAutoRestart() {
    _shouldRestartWakeWordListening = false;
  }

  void resumeWakeWordAutoRestart() {
    _shouldRestartWakeWordListening = true;
    if (!_isListening) startWakeWordListening();
  }

  // FIX: Use dynamic for error type to avoid build errors
  void _onError(dynamic error, int sessionId) {
    debugPrint('[WakeWordService] onError callback. sessionId=$sessionId, currentSessionId=$_currentSessionId, error=$error');
    if (sessionId != _currentSessionId) {
      debugPrint('[WakeWordService] onError callback: ignoring stale sessionId=$sessionId');
      return;
    }
    _isListening = false;
    if (_stopCompleter != null && !_stopCompleter!.isCompleted) {
      _stopCompleter!.complete();
    }
    // Auto-restart if allowed
    if (wakeWordShouldBeActive && (shouldAllowWakeWordRestart?.call() ?? true)) {
      Future.delayed(const Duration(milliseconds: 400), () {
        if (!_isListening) startWakeWordListening();
      });
    }
  }

  void _onResult(SpeechRecognitionResult result, int sessionId) {
    if (sessionId != _currentSessionId) return;
    debugPrint('[WakeWordService] onResult callback. sessionId=$sessionId, currentSessionId=$_currentSessionId, _wakeWordDetected=$_wakeWordDetected  result=$result ');
    if (_wakeWordDetected) return;
    final transcript = result.recognizedWords.toLowerCase();
    if (!_wakeWordDetected && wakeWords.any((w) => transcript.contains(w))) {
      debugPrint('Setting _wakeWordDetected to true for sessionId=$sessionId');
      _wakeWordDetected = true;
      debugPrint('[WakeWordService] Wake word detected: $transcript');
      // Stop listening immediately to prevent further wake word detection
      _isListening = false;
      _questionTimeoutTimer?.cancel(); // Cancel any pending question timeout
      //_stopCompleter?.complete(); // Complete the stop completer if waiting
      debugPrint('[WakeWordService] Stopping wake word listening after detection.');
      pauseWakeWordAutoRestart(); // Prevent auto-restart until we handle the question
      stopWakeWordListening(); // <-- await if possible, or just call it
      debugPrint('[WakeWordService] Wake word listening stopped. Firing onWakeWord callback');
      onWakeWord?.call(transcript);    
    }
  }

  Future<void> startQuestionListening({int retryCount = 0}) async {
    debugPrint('[QuestionService] startQuestionListening called.');
    if (!_wakeWordDetected) return;
    debugPrint('[QuestionService] Forcing _isListening = false before question recognizer.');
    _isListening = false;
    debugPrint('[QuestionService] Stopping any previous recognizer before initializing for question...');
    await _questionSpeech.stop();
    await Future.delayed(const Duration(milliseconds: 400));
    debugPrint('[QuestionService] Initializing speech-to-text for question listening... (retryCount: $retryCount)');

    bool _hasRetried = false;
    bool _hasProcessedResult = false;
    Timer? _localTimeoutTimer;

    bool available = await _questionSpeech.initialize(
      onStatus: (status) {
        debugPrint('[QuestionService] Speech status (question): $status');
        if (status == 'done') {
          debugPrint('[QuestionService] Question recognizer done. Resetting wakeWordDetected to false.');
          _wakeWordDetected = false;
          _questionTimeoutTimer?.cancel();
          _localTimeoutTimer?.cancel();
          if (!_hasProcessedResult) {
            debugPrint('[QuestionService] onStatus: No result processed, restarting wake word listening.');
            resumeWakeWordAutoRestart();
          }
        }
      },
      onError: (error) async {
        debugPrint('[QuestionService] Speech error (question): $error');
        final msg = error.errorMsg.toLowerCase();
        _questionTimeoutTimer?.cancel();
        _localTimeoutTimer?.cancel();
        if (!_hasProcessedResult && (msg.contains('no speech detected') || msg.contains('error_no_match'))) {
          if (!_hasRetried && retryCount < 1) {
            debugPrint('[QuestionService] No speech detected, retrying question listening (attempt ${retryCount + 1})...');
            _hasRetried = true;
            await Future.delayed(const Duration(milliseconds: 400));
            startQuestionListening(retryCount: retryCount + 1);
          } else {
            debugPrint('[QuestionService] No speech detected after retry. Announcing and restarting wake word listening.');
            announce("I didn't hear anything. Try again?");
            _wakeWordDetected = false;
            resumeWakeWordAutoRestart();
          }
        } else {
          debugPrint('[QuestionService] Max retries reached or other error. Announcing error and restarting wake word listening.');
          announce("Speech recognition error. Try again?");
          _wakeWordDetected = false;
          resumeWakeWordAutoRestart();
        }
      },
    );
    if (!available) {
      debugPrint('[QuestionService] ERROR: Speech-to-text initialization failed. Question listening not started. Setting wakeWordDetected to false.');
      _wakeWordDetected = false;
      resumeWakeWordAutoRestart();
      return;
    }
    debugPrint('[QuestionService] Speech-to-text initialized. Listening for question...');

    // Timeout if no speech detected in 10 seconds
    _localTimeoutTimer = Timer(const Duration(seconds: 10), () {
      if (!_hasProcessedResult) {
        debugPrint('[QuestionService] Timeout: No speech detected in 10s. Announcing and restarting wake word listening.');
        announce("I didn't hear anything. Try again?");
        _wakeWordDetected = false;
        _questionSpeech.stop();
        resumeWakeWordAutoRestart();
      }
    });

    await _questionSpeech.listen(
      onResult: (result) async {
        if (_hasProcessedResult) return;
        final question = result.recognizedWords.trim();
        debugPrint('[QuestionService] Question listen transcript: $question');
        _lastPartialQuestion = question;
        // Reset the timer on every result
        _questionTimeoutTimer?.cancel();
        _questionTimeoutTimer = Timer(const Duration(seconds: 2), () async {
          if (_hasProcessedResult) return;
          debugPrint('[QuestionService] No speech detected for 2s, stopping question recognizer.');
          await _questionSpeech.stop();
          if (_lastPartialQuestion.isNotEmpty && onQuestion != null) {
            debugPrint('[QuestionService] Timeout: treating last partial as final question: $_lastPartialQuestion. Setting wakeWordDetected to false.');
            _hasProcessedResult = true;
            onQuestion!(_lastPartialQuestion);
            _wakeWordDetected = false;
            _localTimeoutTimer?.cancel();
            resumeWakeWordAutoRestart();
          } else if (_lastPartialQuestion.isEmpty) {
            debugPrint('[QuestionService] Timeout: no speech detected, nothing to process. Setting wakeWordDetected to false.');
            _hasProcessedResult = true;
            _wakeWordDetected = false;
            _localTimeoutTimer?.cancel();
            resumeWakeWordAutoRestart();
          }
        });

        if (result.finalResult && question.isNotEmpty && onQuestion != null) {
          debugPrint('[QuestionService] Final question recognized, firing onQuestion callback. Setting wakeWordDetected to false.');
          _questionTimeoutTimer?.cancel();
          _hasProcessedResult = true;
          onQuestion!(question);
          _wakeWordDetected = false;
          _localTimeoutTimer?.cancel();
          resumeWakeWordAutoRestart();
        } else {
          debugPrint('[QuestionService] Partial result received, not firing onQuestion.');
        }
      },
      listenFor: const Duration(seconds: 15),
      pauseFor: const Duration(seconds: 3),
      partialResults: true,
      localeId: 'en_US',
      cancelOnError: true,
    );
  }

  void stop() {
    wakeWordShouldBeActive = false;
    _isListening = false;
    debugPrint('Stop() Setting wakeWordShouldBeActive to false');
    _wakeWordDetected = false;
    _currentSessionId = -1; // Invalidate all previous sessions
    _wakeWordSpeech.stop();
    _questionSpeech.stop();
  }

  bool get isListening => _isListening;
  bool get wakeWordDetected => _wakeWordDetected;
}
